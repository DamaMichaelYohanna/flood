IoT Architecture & Measurement Formula

Academic demonstration specifications & NodeMCU ESP8266 firmware

Ultrasonic Water Height Calculation Formula

1. Distance to Water Surface = (Echo Pulse Duration in µs × 0.0343 cm/µs) / 2

2. Water Level (m) = Total Sensor Mounting Height (e.g. 3.00 m) - Measured Distance (m)

Hardware to Web Telemetry Flow

1. JSN-SR04T Waterproof Probe sends 40kHz sonic burst
2. NodeMCU ESP8266 Calculates height & packages JSON
3. Wi-Fi / HTTP API POST payload to REST or Firebase
4. FloodGuard UI Real-time alerts, trends & gauge
NodeMCU ESP8266 Arduino C++ Firmware (Ready to Flash) 
#include <ESP8266WiFi.h>
#include <ESP8266HTTPClient.h>
#include <WiFiClientSecure.h>

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// Replace with your Vercel URL (Note the https:// and /api/readings)
const char* serverUrl = "https://YOUR_VERCEL_APP.vercel.app/api/readings";

#define TRIG_PIN D1  // GPIO5 (Trigger)
#define ECHO_PIN D2  // GPIO4 (Echo)
const float SENSOR_MOUNT_HEIGHT = 3.00; // Sensor height from riverbed (m)

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected! NodeMCU Online.");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  // Emit ultrasonic pulse
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000); // 30ms timeout (~5m max)
  if (duration > 0) {
    float distanceM = (duration * 0.0343) / 2.0 / 100.0;
    float waterLevel = SENSOR_MOUNT_HEIGHT - distanceM;
    if (waterLevel < 0.0) waterLevel = 0.0;

    Serial.printf("Water Level: %.2f m | Distance: %.2f m\n", waterLevel, distanceM);

    // Send HTTPS POST to Vercel Serverless API
    if (WiFi.status() == WL_CONNECTED) {
      WiFiClientSecure client;
      client.setInsecure(); // Required on ESP8266 for HTTPS without root certs

      HTTPClient http;
      if (http.begin(client, serverUrl)) {
        http.addHeader("Content-Type", "application/json");

        String json = "{\"stationId\":\"FLOOD-001\","
                      "\"waterLevel\":" + String(waterLevel, 2) + ","
                      "\"deviceOnline\":true,"
                      "\"sensorConnected\":true,"
                      "\"signalStrength\":" + String(WiFi.RSSI()) + "}";

        int httpCode = http.POST(json);
        Serial.printf("HTTP Code: %d\n", httpCode);
        http.end();
      }
    }
  } else {
    Serial.println("Echo timeout from JSN-SR04T probe.");
  }
  delay(3000); // Sample every 3 seconds
}
